Pattern for all four: build/train a structure or model -> serialize with both pickle and joblib -> deserialize -> print/predict.

Q31 — List of 10 student names

import pickle

import joblib

# (a) Create data structure

students = ["Aarav", "Diya", "Krupesh", "Meera", "Rohan",

"Isha", "Kabir", "Ananya", "Yash", "Priya"]

# (b) Serialize with pickle

with open("students.pkl", "wb") as f:

pickle.dump(students, f)

# Serialize with joblib

joblib.dump(students, "students.joblib")

# Deserialize with pickle

with open("students.pkl", "rb") as f:

loaded_pkl = pickle.load(f)

# Deserialize with joblib

loaded_joblib = joblib.load("students.joblib")

# (c) Print

print("From pickle:", loaded_pkl)

print("From joblib:", loaded_joblib)

Q32 — Classification on Iris dataset

import pickle

import joblib

from sklearn.datasets import load_iris

from sklearn.tree import DecisionTreeClassifier

# (a) Train a small model

iris = load_iris()

X, y = iris.data, iris.target

model = DecisionTreeClassifier()

model.fit(X, y)

# (b) Serialize

with open("iris_model.pkl", "wb") as f:

pickle.dump(model, f)

joblib.dump(model, "iris_model.joblib")

# Deserialize

with open("iris_model.pkl", "rb") as f:

model_pkl = pickle.load(f)

model_joblib = joblib.load("iris_model.joblib")

# (c) Predict + print

new_sample = [[5.1, 3.5, 1.4, 0.2]]

pred = model_pkl.predict(new_sample)

print("Predicted species (pickle model):", iris.target_names[pred[0]])

print("Predicted species (joblib model):", iris.target_names[model_joblib.predict(new_sample)[0]])

Q34 — Classification on Placement dataset (CGPA, IQ → Placed/Not Placed)

import pickle

"Placed": [0, 0, 0, 1, 0, 1, 0, 1]

})

y = data["Placed"]

model = LogisticRegression()

model.fit(X, y)

joblib.dump(model, "placement_model.joblib")

Exam tip: The examiner mainly checks that you use open(...,"wb")/"rb" correctly for pickle vs. joblib.dump/load (no manual file handling needed for joblib), and that you can predict on a new sample afterward.

Topic 4: API Development with FastAPI (Q65–75)

Every question needs the same five things: (1) GET all + DELETE by id, raising HTTPException if not found; (2) a computed field + a GET /sorted endpoint; (3) POST with Pydantic validation, raising the specified status code on duplicate/invalid; (4) verify everything in Swagger UI at http://127.0.0.1:8000/docs; (5) use Path(..., description=..., example=...) on path parameters.

Run any file below with: uvicorn filename:app --reload, then open /docs in your browser.

Q65 — books.json

Fully explained template — this is the one to memorize; every other dataset follows the same skeleton.

from fastapi import FastAPI, HTTPException, Path

from pydantic import BaseModel, Field

import json, os

app = FastAPI()

FILE = "books.json"

def load_data():

if not os.path.exists(FILE):

return []

with open(FILE, "r") as f:

return json.load(f)

def save_data(data):

with open(FILE, "w") as f:

json.dump(data, f, indent=4)

class Book(BaseModel):

id: int

title: str = Field(..., description="Title of the book", example="Atomic Habits")

author: str = Field(..., description="Author of the book", example="James Clear")

price: float = Field(..., gt=0, description="Price of the book", example=499.0)

available: bool = Field(..., description="Whether the book is in stock", example=True)

@app.get("/books")

def get_books():

return load_data()

@app.delete("/books/{book_id}")

def delete_book(book_id: int = Path(..., description="ID of the book to delete", example=1)):

books = load_data()

for b in books:

if b["id"] == book_id:

books.remove(b)

save_data(books)

return {"message": f"Book {book_id} deleted"}

raise HTTPException(status_code=404, detail="Book not found")

@app.get("/books/sorted")

def get_books_sorted():

books = load_data()

for b in books:

b["discount"] = round(b["price"] * 0.10, 2)

books.sort(key=lambda x: x["discount"], reverse=True)

return books

@app.post("/books")

def add_book(book: Book):

books = load_data()

if any(b["id"] == book.id for b in books):

raise HTTPException(status_code=400, detail="Book already exists")

books.append(book.dict())

save_data(books)

print("Book added successfully:", book.title)

return {"message": "Book added successfully", "book": book}
